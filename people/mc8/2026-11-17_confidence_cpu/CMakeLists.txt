cmake_minimum_required(VERSION 3.20)
project(mc8_confidence_cpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(APPLE AND EXISTS "${CMAKE_OSX_SYSROOT}/usr/include/c++/v1")
    # This laptop's CLT compiler searches a partial toolchain header tree.
    # Use the complete libc++ headers shipped in its installed SDK.
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-isystem${CMAKE_OSX_SYSROOT}/usr/include/c++/v1>")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(.local/llama-src llama)
add_executable(confidence-vlm native_vlm.cpp)
target_include_directories(confidence-vlm PRIVATE .local/llama-src/vendor)
target_link_libraries(confidence-vlm PRIVATE llama mtmd)
set_target_properties(confidence-vlm PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
